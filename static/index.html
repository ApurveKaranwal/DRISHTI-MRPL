viscosity_cSt = 15.0      # Kinematic viscosity

velocity_m_s = (flow_rate_m3_h / 3600.0) / (3.14159 * (diameter_m**2) / 4.0)
reynolds = (velocity_m_s * diameter_m) / (viscosity_cSt * 1e-6)
f = 0.02305 # Swamee-Jain friction factor
delta_p_bar = (f * (length_m/diameter_m) * 0.5 * density_kg_m3 * (velocity_m_s**2)) / 100000.0
print(f"Total Pressure Drop: {delta_p_bar:.3f} bar")
